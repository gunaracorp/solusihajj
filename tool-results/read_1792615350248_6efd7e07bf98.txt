    19→/* ================= UI PREFS — dark mode & mode lansia (persist, hydration-safe) ================= */
    20→type Prefs = { dark: boolean; elder: boolean };
    21→
    22→const PREFS_EVT = "shj-prefs";
    23→let prefsCache = "{}";
    24→
    25→function prefsSubscribe(cb: () => void): () => void {
    26→  window.addEventListener(PREFS_EVT, cb);
    27→  window.addEventListener("storage", cb);
    28→  return () => {
    29→    window.removeEventListener(PREFS_EVT, cb);
    30→    window.removeEventListener("storage", cb);
    31→  };
    32→}
    33→
    34→function prefsSnapshot(): string {
    35→  const raw = window.localStorage.getItem("shj_prefs") ?? "{}";
    36→  if (raw !== prefsCache) prefsCache = raw;
    37→  return prefsCache;
    38→}
    39→
    40→const prefsServerSnapshot = "{}";
    41→
    42→function parsePrefs(json: string): Prefs {
    43→  try {
    44→    const p = JSON.parse(json) as Partial<Prefs>;
    45→    return { dark: !!p.dark, elder: !!p.elder };
    46→  } catch {
    47→    return { dark: false, elder: false };
    48→  }
    49→}
    50→
    51→function usePrefs(): [Prefs, (p: Prefs) => void] {
    52→  const json = useSyncExternalStore(prefsSubscribe, prefsSnapshot, () => prefsServerSnapshot);
    53→  const prefs = parsePrefs(json);
    54→  // Sinkronkan class ke <html> setiap kali prefs berubah (eksternal: DOM)
    55→  useEffect(() => {
    56→    document.documentElement.classList.toggle("dark", prefs.dark);
    57→    document.documentElement.classList.toggle("elder", prefs.elder);
    58→  }, [prefs.dark, prefs.elder]);
    59→  const set = (p: Prefs) => {
    60→    try {
    61→      window.localStorage.setItem("shj_prefs", JSON.stringify(p));
    62→    } catch {
    63→      /* noop */
    64→    }
    65→    prefsCache = JSON.stringify(p);
    66→    window.dispatchEvent(new Event(PREFS_EVT));
    67→  };
    68→  return [prefs, set];
    69→}
