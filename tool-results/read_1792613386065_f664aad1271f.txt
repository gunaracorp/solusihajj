    19→    19→/* ================= UI PREFS — dark mode & mode lansia (persist, hydration-safe) ================= */
    20→    20→type Prefs = { dark: boolean; elder: boolean };
    21→    21→
    22→    22→const PREFS_EVT = "shj-prefs";
    23→    23→let prefsCache = "{}";
    24→    24→
    25→    25→function prefsSubscribe(cb: () => void): () => void {
    26→    26→  window.addEventListener(PREFS_EVT, cb);
    27→    27→  window.addEventListener("storage", cb);
    28→    28→  return () => {
    29→    29→    window.removeEventListener(PREFS_EVT, cb);
    30→    30→    window.removeEventListener("storage", cb);
    31→    31→  };
    32→    32→}
    33→    33→
    34→    34→function prefsSnapshot(): string {
    35→    35→  const raw = window.localStorage.getItem("shj_prefs") ?? "{}";
    36→    36→  if (raw !== prefsCache) prefsCache = raw;
    37→    37→  return prefsCache;
    38→    38→}
    39→    39→
    40→    40→const prefsServerSnapshot = "{}";
    41→    41→
    42→    42→function parsePrefs(json: string): Prefs {
    43→    43→  try {
    44→    44→    const p = JSON.parse(json) as Partial<Prefs>;
    45→    45→    return { dark: !!p.dark, elder: !!p.elder };
    46→    46→  } catch {
    47→    47→    return { dark: false, elder: false };
    48→    48→  }
    49→    49→}
    50→    50→
    51→    51→function usePrefs(): [Prefs, (p: Prefs) => void] {
    52→    52→  const json = useSyncExternalStore(prefsSubscribe, prefsSnapshot, () => prefsServerSnapshot);
    53→    53→  const prefs = parsePrefs(json);
    54→    54→  // Sinkronkan class ke <html> setiap kali prefs berubah (eksternal: DOM)
    55→    55→  useEffect(() => {
    56→    56→    document.documentElement.classList.toggle("dark", prefs.dark);
    57→    57→    document.documentElement.classList.toggle("elder", prefs.elder);
    58→    58→  }, [prefs.dark, prefs.elder]);
    59→    59→  const set = (p: Prefs) => {
    60→    60→    try {
    61→    61→      window.localStorage.setItem("shj_prefs", JSON.stringify(p));
    62→    62→    } catch {
    63→    63→      /* noop */
    64→    64→    }
    65→    65→    prefsCache = JSON.stringify(p);
    66→    66→    window.dispatchEvent(new Event(PREFS_EVT));
    67→    67→  };
    68→    68→  return [prefs, set];
    69→    69→}
