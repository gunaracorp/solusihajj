    30→/* ================= BUDGET FILTER HELPERS ================= */
    31→function budgetMatch(price: number, budget: string | null): boolean {
    32→  switch (budget) {
    33→    case "lt30": return price < 30_000_000;
    34→    case "30-40": return price >= 30_000_000 && price <= 40_000_000;
    35→    case "40-50": return price > 40_000_000 && price <= 50_000_000;
    36→    case "gt50": return price > 50_000_000;
    37→    default: return true;
    38→  }
    39→}

    43→  ramadhan: "Ramadhan",

    45→  "plus-dubai": "Plus Dubai",
    46→};

    48→/* ================= KATALOG PAKET ================= */
    49→export function CatalogPage({ query }: { query: URLSearchParams }) {
    50→  const [kota, setKota] = useState(query.get("kota") ?? "");
    51→  const [tipe, setTipe] = useState(query.get("type") ?? "");
    52→  const [budget, setBudget] = useState(query.get("budget") ?? "");
    53→  const [bulan, setBulan] = useState(query.get("bulan") ?? "");
    54→  const [sort, setSort] = useState("harga-naik");
    55→  const [filtersOpen, setFiltersOpen] = useState(false);
    56→
    57→  const filtered = useMemo(() => {
    58→    let list = packageList.filter((p) => {
    59→      if (kota && p.departCity !== kota) return false;
    60→      if (tipe && p.type !== tipe) return false;
    61→      if (!budgetMatch(p.basePrice, budget)) return false;
    62→      if (bulan && !p.departWindow.toLowerCase().includes(bulan.toLowerCase())) return false;
    63→      return true;
    64→    });
    65→    list = [...list].sort((a, b) => {
    66→      switch (sort) {
    67→        case "harga-naik": return a.basePrice - b.basePrice;
    68→        case "harga-turun": return b.basePrice - a.basePrice;
    69→        case "rating": return b.rating - a.rating;
    70→        case "durasi": return a.durationDays - b.durationDays;
    71→        default: return 0;
    72→      }
    73→    });
    74→    return list;
    75→  }, [kota, tipe, budget, bulan, sort]);
    76→
    77→  const activeFilters = [kota, tipe, budget, bulan].filter(Boolean).length;
    78→
    79→  const filterBody = (
    80→    <div className="space-y-5">
    81→      <div>
    82→        <p className="mb-2 text-xs font-bold uppercase tracking-wide text-ink-muted">Tipe Paket</p>
    83→        <div className="flex flex-wrap gap-2">
    84→          {["", "reguler", "ramadhan", "plus-turki", "plus-dubai"].map((t) => (
    85→            <button
    86→              key={t}
    87→              onClick={() => setTipe(t)}
    88→              className={`min-h-10 rounded-full border px-4 text-sm font-semibold transition-colors ${
    89→                tipe === t ? "border-brand bg-brand text-ivory" : "border-line-warm bg-white text-ink hover:border-brand/40"
    90→              }`}
    91→            >
    92→              {t === "" ? "Semua" : TYPE_LABELS[t]}
    93→            </button>
    94→          ))}
    95→        </div>
    96→      </div>
    97→      <div>
    98→        <p className="mb-2 text-xs font-bold uppercase tracking-wide text-ink-muted">Kota Keberangkatan</p>
    99→        <select value={kota} onChange={(e) => setKota(e.target.value)} className="h-12 w-full rounded-xl border border-line-warm bg-white px-3 text-[15px] outline-none focus:border-brand" aria-label="Filter kota">
   100→          <option value="">Semua kota</option>
   101→          {[...new Set(packageList.map((p) => p.departCity))].map((c) => (
   102→            <option key={c} value={c}>{c}</option>
   103→          ))}
   104→        </select>
   105→      </div>
   106→      <div>
   107→        <p className="mb-2 text-xs font-bold uppercase tracking-wide text-ink-muted">Budget / Orang</p>
   108→        <select value={budget} onChange={(e) => setBudget(e.target.value)} className="h-12 w-full rounded-xl border border-line-warm bg-white px-3 text-[15px] outline-none focus:border-brand" aria-label="Filter budget">
   109→          <option value="">Semua budget</option>
   110→          <option value="lt30">&lt; Rp 30 jt</option>
   111→          <option value="30-40">Rp 30 – 40 jt</option>
   112→          <option value="40-50">Rp 40 – 50 jt</option>
   113→          <option value="gt50">&gt; Rp 50 jt</option>
   114→        </select>
   115→      </div>
   116→      <div>
   117→        <p className="mb-2 text-xs font-bold uppercase tracking-wide text-ink-muted">Bulan Keberangkatan</p>
   118→        <select value={bulan} onChange={(e) => setBulan(e.target.value)} className="h-12 w-full rounded-xl border border-line-warm bg-white px-3 text-[15px] outline-none focus:border-brand" aria-label="Filter bulan">
   119→          <option value="">Kapan pun</option>
   120→          {["Februari", "Maret", "April", "Mei", "Juni", "Juli", "Agustus", "September", "Oktober", "Desember"].map((m) => (
   121→            <option key={m} value={m}>{m} 2026</option>
   122→          ))}
   123→        </select>
   124→      </div>
   125→      {activeFilters > 0 && (
   126→        <button
   127→          onClick={() => { setKota(""); setTipe(""); setBudget(""); setBulan(""); }}
   128→          className="inline-flex min-h-10 w-full items-center justify-center gap-1.5 rounded-xl border border-line-warm text-sm font-semibold text-ink-muted hover:bg-muted"
   129→        >
   130→          <X size={14} /> Reset {activeFilters} filter
   131→        </button>
   132→      )}
   133→    </div>
   134→  );
   135→
   136→  return (
   137→    <div className="mx-auto max-w-7xl px-4 py-8 md:px-8 md:py-12">
   138→      <Breadcrumb items={[{ label: "Beranda", to: "/" }, { label: "Paket Umrah" }]} />
   139→      <SectionHeading
   140→        kicker="Katalog"
   141→        title={`${filtered.length} Paket Umrah Terverifikasi`}
   142→        sub="Semua harga adalah harga all-in per orang (quad share). Nama hotel dan jarak meter ke Haram tercantum apa adanya — bukan klaim."
   143→      />
   144→      <div className="mt-8 grid gap-6 lg:grid-cols-[280px_1fr]">
   145→        {/* Sidebar filter desktop */}
   146→        <aside className="hidden h-fit rounded-2xl border border-line-warm bg-card p-5 shadow-card lg:block" aria-label="Filter paket">
   147→          {filterBody}
   148→        </aside>
   149→        <div>
   150→          <div className="mb-4 flex flex-wrap items-center justify-between gap-3">
   151→            <button
   152→              onClick={() => setFiltersOpen(!filtersOpen)}
   153→              className="inline-flex min-h-11 items-center gap-2 rounded-xl border border-brand px-4 font-bold text-brand lg:hidden"
   154→              aria-expanded={filtersOpen}
   155→            >
   156→              <SlidersHorizontal size={16} /> Filter {activeFilters > 0 && `(${activeFilters})`}
   157→            </button>
   158→            <label className="flex items-center gap-2 text-sm text-ink-muted">
   159→              Urutkan:
   160→              <select value={sort} onChange={(e) => setSort(e.target.value)} className="h-11 rounded-xl border border-line-warm bg-white px-3 text-[15px] font-semibold text-ink outline-none focus:border-brand" aria-label="Urutkan hasil">
   161→                <option value="harga-naik">Harga terendah</option>
   162→                <option value="harga-turun">Harga tertinggi</option>
   163→                <option value="rating">Rating tertinggi</option>
   164→                <option value="durasi">Durasi tersingkat</option>
   165→              </select>
   166→            </label>
   167→          </div>
   168→          {filtersOpen && (
   169→            <div className="mb-4 rounded-2xl border border-line-warm bg-card p-5 shadow-card lg:hidden">{filterBody}</div>
   170→          )}
   171→          {filtered.length === 0 ? (
   172→            <div className="rounded-2xl border border-dashed border-line-warm bg-card p-10 text-center">
   173→              <PackageX className="mx-auto mb-3 text-ink-muted" size={36} />
   174→              <p className="font-bold text-ink">Tidak ada paket yang cocok</p>
   175→              <p className="mt-1 text-sm text-ink-muted">Coba longgarkan filter — atau tanyakan via WhatsApp, kami carikan yang paling mendekati.</p>
   176→            </div>
   177→          ) : (
   178→            <div className="grid gap-5 sm:grid-cols-2 xl:grid-cols-3">
   179→              {filtered.map((pkg) => (
   180→                <PackageCard key={pkg.slug} pkg={pkg} ppiu={getPpiu(pkg.ppiuSlug)} />
   181→              ))}
   182→            </div>
   183→          )}
   184→          <div className="mt-8 rounded-2xl bg-brand-soft p-5 text-center">
   185→            <p className="text-[15px] text-brand-deep">
   186→              Tidak menemukan tanggal yang pas? <a href={waLink(waMessages.general)} target="_blank" rel="noopener noreferrer" className="font-bold underline">Tanyakan via WhatsApp</a> — kami bantu cari dari 10 PPIU partner.
   187→            </p>
   188→          </div>
   189→        </div>
   190→      </div>
   191→    </div>
   192→  );
   193→}
